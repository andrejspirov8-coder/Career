Group by tier\n    by_tier = defaultdict(list)\n    by_company = defaultdict(list)\n    by_response = defaultdict(int)\n    \n    for row in rows:\n        score = float(row.get(\"primary_score\") or 0)\n        tier = row.get(\"tier\") or tier_from_score(score)\n        company = row.get(\"company\", \"Unknown\")\n        status = row.get(\"status\", \"\")\n        response = row.get(\"response_status\", \"no_response\")\n        \n        by_tier[tier].append(row)\n        by_company[company].append(row)\n        by_response[response] += 1\n    \n    # Compute response rates\n    tier_stats = {}\n    for tier, profiles in by_tier.items():\n        responses = sum(1 for p in profiles if p.get(\"response_status\") in [\"accepted\", \"messaged\", \"viewed\"])\n        rate = 100 * responses / len(profiles) if profiles else 0\n        tier_stats[tier] = {\"sent\": len(profiles), \"responses\": responses, \"rate\": rate}\n    \n    # Generate report\n    report = f\"\"\"# Recruiter Quarterly Report\n\nGenerated: {datetime.now().isoformat()}\n\n## Summary\n\n- Total contacts: {len(rows)}\n- Total responses: {by_response['accepted'] + by_response['messaged']}\n- Overall response rate: {100 * (by_response['accepted'] + by_response['messaged']) / len(rows):.1f}%\n\n## By Tier\n\n\"\"\"\n    \n    for tier in sorted(tier_stats.keys()):\n        stats = tier_stats[tier]\n        report += f\"- {tier}: {stats['sent']} sent, {stats['responses']} responses ({stats['rate']:.0f}%)\\n\"\n    \n    report += f\"\\n## By Company\\n\\n\"\n    for company in sorted(by_company.keys()):\n        profiles = by_company[company]\n        responses = sum(1 for p in profiles if p.get(\"response_status\") in [\"accepted\", \"messaged\"])\n        rate = 100 * responses / len(profiles) if profiles else 0\n        report += f\"- {company}: {len(profiles)} sent, {responses} responses ({rate:.0f}%)\\n\"\n    \n    OUTPUT_REPORT.write_text(report)\n    print(f\"Report written to {OUTPUT_REPORT}\")\n\nif __name__ == \"__main__\":\n    analyze_recruiter_data()\n```\n\n### Run Weekly\n\n```bash\ncd job-search\npython tools/recruiter_quarterly_report.py\n\n# Output: pipeline/recruiter_quarterly_report.md\n```\n\n---\n\n## Summary: Feedback Loop\n\n**Cycle:**\n\n1. **Scout & Score** → Send 5–10 connections per day (via recruiter_orchestrate.py daily)\n2. **Track Responses** → Log reply status weekly (via spreadsheet or LinkedIn notifications)\n3. **Analyze** → Run quarterly_report.py monthly, extract patterns\n4. **Adjust** → Raise/lower score thresholds, deprioritize low-response sources, boost keyword signals\n5. **Repeat** → Next cycle uses refined config\n\n**Success metrics:** Response rate ≥25%, conversion to interviews ≥8%, top-tier response ≥40%.\n"
